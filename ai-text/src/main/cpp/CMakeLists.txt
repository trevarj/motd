function(ai_source_pin name output)
    set(matches "")
    foreach(line IN LISTS MOTD_LOCK)
        if(line MATCHES "^${name}=.+$")
            list(APPEND matches "${line}")
        endif()
    endforeach()
    list(LENGTH matches count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "source.lock must pin ${name} exactly once")
    endif()
    string(REGEX REPLACE "^[^=]+=" "" value "${matches}")
    set(${output} "${value}" PARENT_SCOPE)
endfunction()
