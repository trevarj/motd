cmake_minimum_required(VERSION 3.31.6)
project(motd_whisper LANGUAGES C CXX)

if(NOT ANDROID OR NOT ANDROID_ABI MATCHES "^(arm64-v8a|x86_64)$")
    message(FATAL_ERROR "motd_whisper supports only Android arm64-v8a and x86_64")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

# Debug APKs are the physical-device development build; keep inference optimized.
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    if(NOT DEFINED MOTD_DEBUG_OPTIMIZATION)
        message(FATAL_ERROR "MOTD_DEBUG_OPTIMIZATION must be pinned")
    endif()
    add_compile_options("${MOTD_DEBUG_OPTIMIZATION}")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(WHISPER_USE_SYSTEM_GGML OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(WHISPER_OPENVINO OFF CACHE BOOL "" FORCE)
set(WHISPER_VITISAI OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL_EMBED_KERNELS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_STATIC OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()

get_filename_component(MOTD_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(WHISPER_PINNED_SOURCE "${MOTD_ROOT}/third_party/whisper.cpp/source")
if(NOT EXISTS "${WHISPER_PINNED_SOURCE}/include/whisper.h")
    message(FATAL_ERROR "Pinned whisper.cpp source is missing: ${WHISPER_PINNED_SOURCE}")
endif()

set(WHISPER_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/whisper-source")
file(REMOVE_RECURSE "${WHISPER_SOURCE}")
file(COPY "${WHISPER_PINNED_SOURCE}/" DESTINATION "${WHISPER_SOURCE}" PATTERN ".git" EXCLUDE)
set(MOTD_WHISPER_PATCH_SOURCE "${WHISPER_SOURCE}/src/whisper.cpp")
include("${MOTD_ROOT}/third_party/ai/patches/whisper-allocation-failure.cmake")
unset(MOTD_WHISPER_PATCH_SOURCE)

add_subdirectory("${WHISPER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/whisper" EXCLUDE_FROM_ALL)

add_library(motd_whisper SHARED whisper_jni.cpp)
target_include_directories(motd_whisper PRIVATE
    "${WHISPER_SOURCE}/include"
    "${WHISPER_SOURCE}/ggml/include")
target_link_libraries(motd_whisper PRIVATE whisper)
target_link_options(motd_whisper PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
